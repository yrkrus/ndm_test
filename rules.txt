# ----------------------------------------------------
# Базовый набор AT команд для тестирования
# Формат: Ожидание=Ответ
# \r\n здесь пишется текстом, парсер потом заменит на реальные переносы
# ----------------------------------------------------

# 1. Проверка связи
AT=OK\r\n

# 2. Управление эхом (ATE0 / ATE1). 
# (.), чтобы правило срабатывало и на 0, и на 1
ATE.=OK\r\n

# 3. Информация о модеме. 
# Ответ из нескольких строк, чтобы проверить работу переносов \r\n
ATI=Quectel\r\nEC21\r\nRevision: EC21EFAR06A05M4G\r\n\r\nOK\r\n

# 4. Выбор оператора (COPS)
# (*), чтобы отвечать на AT+COPS?, AT+COPS=0, AT+COPS=1 и т.д.
# Звезда означает "любые символы после COPS"
AT+COPS*=+COPS: 0,0,"MTS",7\r\n\r\nOK\r\n

# 5. Статус SIM-карты (CPIN?)
AT+CPIN?=+CPIN: READY\r\n\r\nOK\r\n

# 6. Ввод PIN-кода (CPIN="1234")
# Звезда в кавычках означает, что мы примем любой PIN-код
AT+CPIN="*"=OK\r\n

# 7. Обработка ошибок
# Если команда не совпала ни с одним правилом выше, 
# сработает этот паттерн (звезда соответствует всей строке) и вернется ERROR
*=ERROR\r\n
